{{ _("Debian and Ubuntu") }}

{{ _("Add Clementine's apt repository to get new versions with the rest of your updates. It has packages for the current releases of Debian and Ubuntu, and Ubuntu-based distributions such as Linux Mint, for 64-bit PCs and ARM.") }}

sudo install -d -m 0755 /etc/apt/keyrings
wget -qO- https://apt.clementine-player.org/clementine-archive-keyring.gpg \
  | sudo tee /etc/apt/keyrings/clementine-archive-keyring.gpg > /dev/null
sudo tee /etc/apt/sources.list.d/clementine.sources > /dev/null <<EOF
Types: deb
URIs: https://apt.clementine-player.org
Suites: $(. /etc/os-release && echo "${UBUNTU_CODENAME:-$VERSION_CODENAME}")
Components: main
Signed-By: /etc/apt/keyrings/clementine-archive-keyring.gpg
EOF
sudo apt update
sudo apt install clementine
